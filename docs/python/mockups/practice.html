from collections import defaultdict def revenue_by_category(items, products): category_of = {p["product_id"]: p["category"] for p in products} totals = defaultdict(float) for line in items: category = category_of.get(line["product_id"]) if category is not None: totals[category] += line["line_total"] ranked = sorted(totals.items(), key=lambda kv: kv[1], reverse=True) return {c: round(v, 2) for c, v in ranked}
